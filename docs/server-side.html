Server-Side Integration

Overview

Server-side processing allows you to handle large datasets efficiently by processing pagination, sorting, and filtering on the server rather than in the browser. This is essential for tables with thousands or millions of records.

Benefits of Server-Side Processing

  • Handle large datasets (millions of records)
  • Faster initial page load
  • Reduced memory usage in browser
  • Real-time data from database
  • Advanced filtering and searching capabilities

Vue.js Setup

Basic Server-Side Component

<template>
  <div>
    <DataTable
      :data="products"
      :columns="tableColumns"
      :config="tableConfig"
      :pagination-info="paginationInfo"
      :loading="loading"
      server-side
      @filter="handleFilter"
      @sort="handleSort"
      @paginate="handlePaginate"
    />
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import DataTable from '@/components/DataTable/DataTable.vue'

// State
const products = ref([])
const loading = ref(false)
const currentFilters = ref({})
const currentSort = ref({ column: null, direction: null })
const paginationInfo = ref({
  currentPage: 1,
  pageSize: 10,
  total: 0,
  totalPages: 0
})

// Table Configuration
const tableColumns = ref([
  {
    key: 'id',
    title: 'ID',
    sortable: true,
    filterable: true,
    filterType: 'text',
    width: '80px'
  },
  {
    key: 'name',
    title: 'Product Name',
    sortable: true,
    filterable: true,
    filterType: 'text'
  },
  {
    key: 'category',
    title: 'Category',
    sortable: true,
    filterable: true,
    filterType: 'select',
    filterOptions: [
      { value: '', label: 'All Categories' },
      { value: 'Electronics', label: 'Electronics' },
      { value: 'Clothing', label: 'Clothing' },
      { value: 'Books', label: 'Books' }
    ]
  },
  {
    key: 'price',
    title: 'Price',
    type: 'currency',
    sortable: true,
    filterable: true,
    filterType: 'text',
    align: 'right'
  },
  {
    key: 'status',
    title: 'Status',
    type: 'badge',
    sortable: true,
    filterable: true,
    filterType: 'select',
    filterOptions: [
      { value: '', label: 'All Status' },
      { value: 'active', label: 'Active' },
      { value: 'inactive', label: 'Inactive' }
    ]
  }
])

const tableConfig = ref({
  pagination: {
    enabled: true,
    pageSize: 10,
    pageSizes: [5, 10, 25, 50, 100]
  },
  filtering: {
    enabled: true,
    global: true,
    column: true,
    additionalFilters: {
      enabled: true
    }
  },
  ui: {
    scrollable: {
      enabled: true,
      maxHeight: '500px'
    }
  }
})

// API Functions
async function fetchData() {
  loading.value = true
  try {
    const params = buildQueryParams()
    const response = await fetch(`/api/products?${params}`)
    const data = await response.json()
    
    products.value = data.data
    paginationInfo.value = {
      currentPage: data.current_page,
      pageSize: data.per_page,
      total: data.total,
      totalPages: data.last_page
    }
  } catch (error) {
    console.error('Error fetching data:', error)
  } finally {
    loading.value = false
  }
}

function buildQueryParams() {
  const params = new URLSearchParams()
  
  // Pagination
  params.append('page', paginationInfo.value.currentPage)
  params.append('per_page', paginationInfo.value.pageSize)
  
  // Sorting
  if (currentSort.value.column) {
    params.append('sort', currentSort.value.column)
    params.append('direction', currentSort.value.direction)
  }
  
  // Filtering
  Object.entries(currentFilters.value).forEach(([key, filter]) => {
    if (filter.value) {
      params.append(`filter[${key}]`, filter.value)
      params.append(`filter_type[${key}]`, filter.type || 'contains')
    }
  })
  
  return params.toString()
}

// Event Handlers
async function handleFilter({ key, value, type }) {
  if (value === '' || value === null || value === undefined) {
    delete currentFilters.value[key]
  } else {
    currentFilters.value[key] = { value, type: type || 'contains' }
  }
  paginationInfo.value.currentPage = 1
  await fetchData()
}

async function handleSort({ column, direction }) {
  currentSort.value = { column, direction }
  await fetchData()
}

async function handlePaginate({ page, pageSize }) {
  paginationInfo.value.currentPage = page
  if (pageSize) {
    paginationInfo.value.pageSize = pageSize
  }
  await fetchData()
}

// Initialize
onMounted(() => {
  fetchData()
})
</script>

Laravel Backend Setup

Controller Implementation

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Product;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function index(Request $request)
    {
        $query = Product::query();
        
        // Apply filters
        $this->applyFilters($query, $request);
        
        // Apply sorting
        $this->applySorting($query, $request);
        
        // Get paginated results
        $perPage = $request->get('per_page', 10);
        $products = $query->paginate($perPage);
        
        return response()->json($products);
    }
    
    private function applyFilters($query, Request $request)
    {
        $filters = $request->get('filter', []);
        $filterTypes = $request->get('filter_type', []);
        
        foreach ($filters as $field => $value) {
            if (empty($value)) continue;
            
            $type = $filterTypes[$field] ?? 'contains';
            
            switch ($type) {
                case 'contains':
                    $query->where($field, 'LIKE', "%{$value}%");
                    break;
                case 'equals':
                    $query->where($field, $value);
                    break;
                case 'starts_with':
                    $query->where($field, 'LIKE', "{$value}%");
                    break;
                case 'ends_with':
                    $query->where($field, 'LIKE', "%{$value}");
                    break;
                case 'greater_than':
                    $query->where($field, '>', $value);
                    break;
                case 'less_than':
                    $query->where($field, '<', $value);
                    break;
                case 'date':
                    $query->whereDate($field, $value);
                    break;
            }
        }
        
        // Handle global search
        if ($globalSearch = $request->get('search')) {
            $query->where(function ($q) use ($globalSearch) {
                $q->where('name', 'LIKE', "%{$globalSearch}%")
                  ->orWhere('description', 'LIKE', "%{$globalSearch}%")
                  ->orWhere('category', 'LIKE', "%{$globalSearch}%");
            });
        }
    }
    
    private function applySorting($query, Request $request)
    {
        $sortField = $request->get('sort');
        $sortDirection = $request->get('direction', 'asc');
        
        if ($sortField) {
            $query->orderBy($sortField, $sortDirection);
        } else {
            // Default sorting
            $query->orderBy('created_at', 'desc');
        }
    }
}

Model Setup

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Product extends Model
{
    use HasFactory;
    
    protected $fillable = [
        'name',
        'description',
        'category',
        'price',
        'stock',
        'status',
        'created_by',
        'updated_by'
    ];
    
    protected $casts = [
        'price' => 'decimal:2',
        'stock' => 'integer',
        'created_at' => 'datetime',
        'updated_at' => 'datetime'
    ];
    
    // Relationships
    public function creator()
    {
        return $this->belongsTo(User::class, 'created_by');
    }
    
    public function updater()
    {
        return $this->belongsTo(User::class, 'updated_by');
    }
}

Advanced Server-Side Features

Additional Filters

Handle additional filters like created_at, updated_by, etc.:

// In your controller
private function applyAdditionalFilters($query, Request $request)
{
    // Handle created_at filter
    if ($createdAt = $request->get('filter.created_at')) {
        $query->whereDate('created_at', $createdAt);
    }
    
    // Handle updated_at filter
    if ($updatedAt = $request->get('filter.updated_at')) {
        $query->whereDate('updated_at', $updatedAt);
    }
    
    // Handle created_by filter
    if ($createdBy = $request->get('filter.created_by')) {
        $query->whereHas('creator', function ($q) use ($createdBy) {
            $q->where('name', 'LIKE', "%{$createdBy}%");
        });
    }
    
    // Handle updated_by filter
    if ($updatedBy = $request->get('filter.updated_by')) {
        $query->whereHas('updater', function ($q) use ($updatedBy) {
            $q->where('name', 'LIKE', "%{$updatedBy}%");
        });
    }
}

Performance Optimization

Performance Tips

  • Add database indexes on frequently filtered/sorted columns
  • Use eager loading for relationships
  • Implement caching for static filter options
  • Consider using database views for complex queries
  • Limit the maximum page size to prevent memory issues
