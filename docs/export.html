Export Features

Overview

The DataTable component provides comprehensive export functionality, allowing users to export table data in multiple formats including CSV, Excel, and JSON. Export respects current filters and can be customized for different use cases.

Supported Formats

  • CSV: Comma-separated values for spreadsheet applications
  • Excel: Native Excel format (.xlsx) with formatting
  • JSON: JavaScript Object Notation for developers
  • Print: Print-friendly table layout

Custom Export Logic

Server-Side Export

<script setup>
// Custom export handler for server-side exports
async function handleExport({ format, filename, data }) {
  console.log('Export requested:', { format, filename })
  
  if (format === 'excel' && data.length > 1000) {
    // For large datasets, use server-side export
    await exportFromServer(format, filename)
  } else {
    // Use client-side export for smaller datasets
    // Component handles this automatically
  }
}

async function exportFromServer(format, filename) {
  try {
    // Build current filters and parameters
    const params = buildQueryParams()
    params.append('export_format', format)
    params.append('export_filename', filename)
    
    // Request export from server
    const response = await fetch(`/api/products/export?${params}`)
    
    if (response.ok) {
      // Download the file
      const blob = await response.blob()
      const url = window.URL.createObjectURL(blob)
      const a = document.createElement('a')
      a.href = url
      a.download = `${filename}.${format === 'excel' ? 'xlsx' : format}`
      document.body.appendChild(a)
      a.click()
      window.URL.revokeObjectURL(url)
      document.body.removeChild(a)
    }
  } catch (error) {
    console.error('Export failed:', error)
  }
}
</script>

Laravel Server-Side Export

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Product;
use Illuminate\Http\Request;
use Maatwebsite\Excel\Facades\Excel;
use App\Exports\ProductsExport;

class ProductController extends Controller
{
    public function export(Request $request)
    {
        $format = $request->get('export_format', 'csv');
        $filename = $request->get('export_filename', 'products-export');
        
        // Build query with filters
        $query = Product::query();
        $this->applyFilters($query, $request);
        $this->applySorting($query, $request);
        
        // Get all data (no pagination for export)
        $products = $query->get();
        
        switch ($format) {
            case 'excel':
                return Excel::download(
                    new ProductsExport($products), 
                    $filename . '.xlsx'
                );
                
            case 'csv':
                return Excel::download(
                    new ProductsExport($products), 
                    $filename . '.csv',
                    \Maatwebsite\Excel\Excel::CSV
                );
                
            case 'json':
                return response()->json($products)
                    ->header('Content-Disposition', 'attachment; filename="' . $filename . '.json"');
                    
            default:
                return response()->json(['error' => 'Unsupported format'], 400);
        }
    }
}

// ProductsExport class
namespace App\Exports;

use Maatwebsite\Excel\Concerns\FromCollection;
use Maatwebsite\Excel\Concerns\WithHeadings;
use Maatwebsite\Excel\Concerns\WithMapping;

class ProductsExport implements FromCollection, WithHeadings, WithMapping
{
    protected $products;
    
    public function __construct($products)
    {
        $this->products = $products;
    }
    
    public function collection()
    {
        return $this->products;
    }
    
    public function headings(): array
    {
        return [
            'ID',
            'Product Name',
            'Category',
            'Price',
            'Stock',
            'Status',
            'Created At'
        ];
    }
    
    public function map($product): array
    {
        return [
            $product->id,
            $product->name,
            $product->category,
            $product->price,
            $product->stock,
            $product->status,
            $product->created_at->format('Y-m-d H:i:s')
        ];
    }
}

// utils/exportUtils.js - Built-in export functions

// CSV Export
export function exportToCSV(data, filename, options = {}) {
  const {
    delimiter = ',',
    includeHeaders = true,
    encoding = 'utf-8'
  } = options
  
  // Convert data to CSV format
  const csvContent = convertToCSV(data, delimiter, includeHeaders)
  
  // Create and download file
  downloadFile(csvContent, `${filename}.csv`, 'text/csv;charset=utf-8;')
}

// Excel Export
export function exportToExcel(data, filename, options = {}) {
  const {
    sheetName = 'Sheet1',
    includeHeaders = true
  } = options
  
  // Create workbook and worksheet
  const workbook = createWorkbook(data, sheetName, includeHeaders)
  
  // Download Excel file
  downloadWorkbook(workbook, `${filename}.xlsx`)
}

// JSON Export
export function exportToJSON(data, filename, options = {}) {
  const {
    pretty = true,
    includeMetadata = false
  } = options
  
  let jsonData = data
  
  if (includeMetadata) {
    jsonData = {
      exportDate: new Date().toISOString(),
      totalRecords: data.length,
      data: data
    }
  }
  
  const jsonString = pretty ? 
    JSON.stringify(jsonData, null, 2) : 
    JSON.stringify(jsonData)
  
  downloadFile(jsonString, `${filename}.json`, 'application/json')
}

  printWindow.print()
}

Best Practices

Performance

  • Use server-side export for large datasets (>10,000 rows)
  • Implement progress indicators for long-running exports
  • Consider pagination for very large exports
  • Cache export results when appropriate
  • Limit concurrent export requests

User Experience

  • Provide clear export format options
  • Include meaningful default filenames
  • Show export progress for large datasets
  • Handle export errors gracefully
  • Respect user's current filters and sorting

Security

  • Validate export permissions on the server
  • Sanitize data before export
  • Limit export frequency to prevent abuse
  • Log export activities for auditing
  • Exclude sensitive fields from exports

Troubleshooting

Export button not showing

  • Check that export.enabled is set to true in config
  • Verify that at least one format is specified in export.formats
  • Ensure the toolbar is enabled

Download not starting

  • Check browser popup blocker settings
  • Verify that the export utilities are properly imported
  • Check browser console for JavaScript errors

Exported data is incomplete

  • Check if includeFiltered is set correctly
  • Verify that all desired columns have exportable: true
  • For server-side: ensure all data is being queried
